    <script language="JavaScript">
      var current_selection;

      // ----------------------------------------------
      function OnTabClicked (tab_id)
      {
        var element = document.getElementById (tab_id);

        if (current_selection == null)
        {
          current_selection = document.getElementById ('round_0');
        }
        current_selection.style.display = "none";

        current_selection = element;
        current_selection.style.display = "inline";

        document.cookie = 'current_tab=' + tab_id;
      }

      // ----------------------------------------------
      function OnLoad ()
      {
        window.OnTabClicked (ReadCookie ('current_tab'));

        setInterval (function(){location.reload ();}, 60*1000);
      }

      // ----------------------------------------------
      function ReadCookie (name)
      {
        return (document.cookie.match ('(^|; )' + name + '=([^;]*)')||0)[2];
      }
    </script>
